API итератора массивов#

Итератор массивов #

Итератор массивов объединяет многие ключевые возможности универсальных функций. Пользовательский код может без сложной реализации поддерживать выходные параметры, сохранение размещения данных в памяти и буферизацию данных с неподходящими выравниванием или типом.

На этой странице описан API итератора. Сам итератор называется NpyIter, а имена функций имеют вид NpyIter_*.

Пользователям этого C API может быть полезно вводное руководство по обходу массивов. Прежде чем писать код обхода на C, часто удобно проверить идею, создав итератор в Python.

Пример обхода#

Лучший способ познакомиться с итератором — посмотреть на его использование в самой кодовой базе NumPy. Например, вот немного измененная версия кода для PyArray_CountNonzero, которая подсчитывает количество ненулевых элементов в массиве.

npy_intp PyArray_CountNonzero(PyArrayObject* self)
{
    /* Nonzero boolean function */
    PyArray_NonzeroFunc* nonzero = PyArray_DESCR(self)->f->nonzero;

    NpyIter* iter;
    NpyIter_IterNextFunc *iternext;
    char** dataptr;
    npy_intp nonzero_count;
    npy_intp* strideptr,* innersizeptr;

    /* Handle zero-sized arrays specially */
    if (PyArray_SIZE(self) == 0) {
        return 0;
    }

    /*
     * Create and use an iterator to count the nonzeros.
     *   flag NPY_ITER_READONLY
     *     - The array is never written to.
     *   flag NPY_ITER_EXTERNAL_LOOP
     *     - Inner loop is done outside the iterator for efficiency.
     *   flag NPY_ITER_NPY_ITER_REFS_OK
     *     - Reference types are acceptable.
     *   order NPY_KEEPORDER
     *     - Visit elements in memory order, regardless of strides.
     *       This is good for performance when the specific order
     *       elements are visited is unimportant.
     *   casting NPY_NO_CASTING
     *     - No casting is required for this operation.
     */
    iter = NpyIter_New(self, NPY_ITER_READONLY|
                             NPY_ITER_EXTERNAL_LOOP|
                             NPY_ITER_REFS_OK,
                        NPY_KEEPORDER, NPY_NO_CASTING,
                        NULL);
    if (iter == NULL) {
        return -1;
    }

    /*
     * The iternext function gets stored in a local variable
     * so it can be called repeatedly in an efficient manner.
     */
    iternext = NpyIter_GetIterNext(iter, NULL);
    if (iternext == NULL) {
        NpyIter_Deallocate(iter);
        return -1;
    }
    /* The location of the data pointer which the iterator may update */
    dataptr = NpyIter_GetDataPtrArray(iter);
    /* The location of the stride which the iterator may update */
    strideptr = NpyIter_GetInnerStrideArray(iter);
    /* The location of the inner loop size which the iterator may update */
    innersizeptr = NpyIter_GetInnerLoopSizePtr(iter);

    nonzero_count = 0;
    do {
        /* Get the inner loop data/stride/count values */
        char* data = *dataptr;
        npy_intp stride = *strideptr;
        npy_intp count = *innersizeptr;

        /* This is a typical inner loop for NPY_ITER_EXTERNAL_LOOP */
        while (count--) {
            if (nonzero(data, self)) {
                ++nonzero_count;
            }
            data += stride;
        }

        /* Increment the iterator to the next inner loop */
    } while(iternext(iter));

    NpyIter_Deallocate(iter);

    return nonzero_count;
}

Пример совместного обхода нескольких массивов#

Вот функция копирования с использованием итератора. Параметр order управляет размещением выделяемого результата в памяти; обычно нужен NPY_KEEPORDER.

PyObject *CopyArray(PyObject *arr, NPY_ORDER order)
{
    NpyIter *iter;
    NpyIter_IterNextFunc *iternext;
    PyObject *op[2], *ret;
    npy_uint32 flags;
    npy_uint32 op_flags[2];
    npy_intp itemsize, *innersizeptr, innerstride;
    char **dataptrarray;

    /*
     * No inner iteration - inner loop is handled by CopyArray code
     */
    flags = NPY_ITER_EXTERNAL_LOOP;
    /*
     * Tell the constructor to automatically allocate the output.
     * The data type of the output will match that of the input.
     */
    op[0] = arr;
    op[1] = NULL;
    op_flags[0] = NPY_ITER_READONLY;
    op_flags[1] = NPY_ITER_WRITEONLY | NPY_ITER_ALLOCATE;

    /* Construct the iterator */
    iter = NpyIter_MultiNew(2, op, flags, order, NPY_NO_CASTING,
                            op_flags, NULL);
    if (iter == NULL) {
        return NULL;
    }

    /*
     * Make a copy of the iternext function pointer and
     * a few other variables the inner loop needs.
     */
    iternext = NpyIter_GetIterNext(iter, NULL);
    innerstride = NpyIter_GetInnerStrideArray(iter)[0];
    itemsize = NpyIter_GetDescrArray(iter)[0]->elsize;
    /*
     * The inner loop size and data pointers may change during the
     * loop, so just cache the addresses.
     */
    innersizeptr = NpyIter_GetInnerLoopSizePtr(iter);
    dataptrarray = NpyIter_GetDataPtrArray(iter);

    /*
     * Note that because the iterator allocated the output,
     * it matches the iteration order and is packed tightly,
     * so we don't need to check it like the input.
     */
    if (innerstride == itemsize) {
        do {
            memcpy(dataptrarray[1], dataptrarray[0],
                                    itemsize * (*innersizeptr));
        } while (iternext(iter));
    } else {
        /* For efficiency, should specialize this based on item size... */
        npy_intp i;
        do {
            npy_intp size = *innersizeptr;
            char *src = dataptrarray[0], *dst = dataptrarray[1];
            for(i = 0; i < size; i++, src += innerstride, dst += itemsize) {
                memcpy(dst, src, itemsize);
            }
        } while (iternext(iter));
    }

    /* Get the result from the iterator object array */
    ret = NpyIter_GetOperandArray(iter)[1];
    Py_INCREF(ret);

    if (NpyIter_Deallocate(iter) != NPY_SUCCEED) {
        Py_DECREF(ret);
        return NULL;
    }

    return ret;
}

Пример отслеживания многомерного индекса#

В этом примере показано, как работать с флагом NPY_ITER_MULTI_INDEX. Для простоты мы предполагаем, что аргумент представляет собой двумерный массив.

int PrintMultiIndex(PyArrayObject *arr) {
    NpyIter *iter;
    NpyIter_IterNextFunc *iternext;
    npy_intp multi_index[2];

    iter = NpyIter_New(
        arr, NPY_ITER_READONLY | NPY_ITER_MULTI_INDEX | NPY_ITER_REFS_OK,
        NPY_KEEPORDER, NPY_NO_CASTING, NULL);
    if (iter == NULL) {
        return -1;
    }
    if (NpyIter_GetNDim(iter) != 2) {
        NpyIter_Deallocate(iter);
        PyErr_SetString(PyExc_ValueError, "Array must be 2-D");
        return -1;
    }
    if (NpyIter_GetIterSize(iter) != 0) {
        iternext = NpyIter_GetIterNext(iter, NULL);
        if (iternext == NULL) {
            NpyIter_Deallocate(iter);
            return -1;
        }
        NpyIter_GetMultiIndexFunc *get_multi_index =
            NpyIter_GetGetMultiIndex(iter, NULL);
        if (get_multi_index == NULL) {
            NpyIter_Deallocate(iter);
            return -1;
        }

        do {
            get_multi_index(iter, multi_index);
            printf("multi_index is [%" NPY_INTP_FMT ", %" NPY_INTP_FMT "]\n",
                   multi_index[0], multi_index[1]);
        } while (iternext(iter));
    }
    if (!NpyIter_Deallocate(iter)) {
        return -1;
    }
    return 0;
}

При вызове с массивом 2x3 вышеприведенный пример печатает:

multi_index is [0, 0]
multi_index is [0, 1]
multi_index is [0, 2]
multi_index is [1, 0]
multi_index is [1, 1]
multi_index is [1, 2]

Типы данных итератора#

Устройство итератора является внутренней деталью реализации; пользовательскому коду доступна лишь неполная структура.

type NpyIter#

Непрозрачный тип указателя на итератор. Доступ к его содержимому возможен только через API итератора.

type NpyIter_Type#

Этот тип делает итератор доступным в Python. Сейчас нет API для доступа из C к значениям итератора, созданного в Python. Итератор, созданный в Python, нужно использовать в Python, и наоборот. Такой API, вероятно, появится в будущей версии.

type NpyIter_IterNextFunc#

Это указатель функции для цикла итерации, возвращаемый NpyIter_GetIterNext.

type NpyIter_GetMultiIndexFunc#

Указатель на функцию получения текущего многомерного индекса итератора, возвращаемый NpyIter_GetGetMultiIndex.

Создание и уничтожение#

NpyIter *NpyIter_New(PyArrayObject *op, npy_uint32 flags, NPY_ORDER order, NPY_CASTING casting, PyArray_Descr *dtype)#

Создаёт итератор для заданного объекта массива NumPy op.

В flags можно передать любую комбинацию глобальных флагов и флагов отдельных операндов, описанных в NpyIter_MultiNew, кроме NPY_ITER_ALLOCATE.

Любое значение перечисления NPY_ORDER можно передать order. Для эффективного обхода лучше всего подходит NPY_KEEPORDER; остальные варианты задают конкретный порядок обхода.

Любое из значений перечисления NPY_CASTING может быть передано в casting. Значения включают NPY_NO_CASTING, NPY_EQUIV_CASTING, NPY_SAFE_CASTING, NPY_SAME_KIND_CASTING и NPY_UNSAFE_CASTING. Чтобы разрешить приведение типов, необходимо также включить копирование или буферизацию.

Если dtype не равно NULL, требуется именно этот тип данных. Если копирование разрешено и данные можно привести к этому типу, будет создана временная копия. Если включён NPY_ITER_UPDATEIFCOPY, при уничтожении итератора данные также будут скопированы обратно с ещё одним приведением типов.

Возвращает NULL в случае ошибки, в противном случае возвращает выделенный итератор.

Для создания итератора, похожего на старый, должен подойти такой вариант.

iter = NpyIter_New(op, NPY_ITER_READWRITE,
                    NPY_CORDER, NPY_NO_CASTING, NULL);

Это подходит, если нужно изменять массив кодом для выровненных значений double, а порядок элементов не важен.

dtype = PyArray_DescrFromType(NPY_DOUBLE);
iter = NpyIter_New(op, NPY_ITER_READWRITE|
                    NPY_ITER_BUFFERED|
                    NPY_ITER_NBO|
                    NPY_ITER_ALIGNED,
                    NPY_KEEPORDER,
                    NPY_SAME_KIND_CASTING,
                    dtype);
Py_DECREF(dtype);
NpyIter *NpyIter_MultiNew(npy_intp nop, PyArrayObject **op, npy_uint32 flags, NPY_ORDER order, NPY_CASTING casting, npy_uint32 *op_flags, PyArray_Descr **op_dtypes)#

Создаёт итератор для трансляции nop объектов массивов, переданных в op, по обычным правилам NumPy.

Любое значение перечисления NPY_ORDER можно передать в order. Для эффективного обхода лучше всего NPY_KEEPORDER; остальные варианты задают конкретный порядок. При NPY_KEEPORDER для запрета обратного обхода вдоль оси передайте флаг NPY_ITER_DONT_NEGATE_STRIDES.

Любое из значений перечисления NPY_CASTING может быть передано в casting. Значения включают NPY_NO_CASTING, NPY_EQUIV_CASTING, NPY_SAFE_CASTING, NPY_SAME_KIND_CASTING и NPY_UNSAFE_CASTING. Чтобы разрешить приведение типов, необходимо также включить копирование или буферизацию.

Если op_dtypes не является NULL, он определяет тип данных или NULL для каждого op[i].

Возвращает NULL в случае ошибки, в противном случае возвращает выделенный итератор.

В flags можно передать следующие флаги, действующие на весь итератор:

NPY_ITER_C_INDEX#

Это приводит к тому, что итератор отслеживает плоский индекс, совпадающий с порядком C. Этот параметр не может быть использован с NPY_ITER_F_INDEX.

NPY_ITER_F_INDEX#

Включает отслеживание итератором плоского индекса, соответствующего порядку Fortran. Этот параметр нельзя использовать вместе с NPY_ITER_C_INDEX.

NPY_ITER_MULTI_INDEX#

Итератор отслеживает многомерный индекс. Это не позволяет объединять оси для увеличения внутренних циклов. Если обход также не буферизуется и линейный индекс не отслеживается, так что допустим вызов NpyIter_RemoveAxis, размер итератора может быть -1, указывая на слишком большой размер. Это возможно при сложной трансляции массивов и приводит к ошибкам при задании диапазона итератора, отключении многомерного индекса или получении функции перехода к следующему элементу. Однако можно удалить оси и пользоваться итератором как обычно, если после удаления размер станет достаточно мал.

NPY_ITER_EXTERNAL_LOOP#

Итератор не выполняет самый внутренний цикл; его обработку должен взять на себя пользователь итератора.

Этот флаг несовместим с NPY_ITER_C_INDEX, NPY_ITER_F_INDEX и NPY_ITER_MULTI_INDEX.

NPY_ITER_DONT_NEGATE_STRIDES#

Влияет на итератор только при значении NPY_KEEPORDER параметра order. По умолчанию при NPY_KEEPORDER итератор меняет направление обхода осей с отрицательным шагом, чтобы двигаться по памяти вперёд. Этот флаг отключает такое изменение. Он нужен, если требуется учитывать расположение осей в памяти, но не менять направление обхода. Так ведёт себя, например, numpy.ravel(a, order='K').

NPY_ITER_COMMON_DTYPE#

Заставляет итератор приводить все операнды к общему типу данных, определяемому правилами продвижения типов ufunc. Копирование или буферизация должны быть разрешены.

Если общий тип данных известен заранее, не используйте этот флаг. вместо этого установите запрашиваемый dtype для всех операндов.

NPY_ITER_REFS_OK#

Указывает, что итератор может принимать и использовать массивы со ссылочными типами (массивы объектов или структурированные массивы, содержащие поля объектного типа). Если этот флаг установлен, вызывающий код обязан проверить, истинно ли значение NpyIter_IterationNeedsAPI(iter); в этом случае освобождать GIL во время итерации нельзя. При работе с известными типами данных NpyIter_GetTransferFlags позволяет быстрее и точнее проверить, требуется ли итератору API из-за буферизации.

NPY_ITER_ZEROSIZE_OK#

Разрешает массивы нулевого размера. Поскольку обычный цикл обхода сам по себе не обрабатывает этот случай, перед входом в него нужно проверить, что IterSize больше нуля. Сейчас проверяются только операнды, а не принудительно заданная форма.

NPY_ITER_REDUCE_OK#

Допускает доступные для записи операнды, у которых есть измерение с нулевым шагом и размером больше 1. Такие операнды должны быть доступны и для чтения, и для записи.

Если включена буферизация, также активируется специальный режим, который при необходимости сокращает длину цикла, чтобы не перезаписать значения, участвующие в свёртке.

Для свёртки с автоматически выделенным выходным массивом необходимо получить ссылку на него через NpyIter_GetOperandArray и перед циклом итерации инициализировать все его элементы нейтральным элементом операции. При буферизованной свёртке также нужно указать флаг NPY_ITER_DELAY_BUFALLOC, а после инициализации выделенного операнда сбросить итератор для подготовки буферов.

NPY_ITER_RANGED#

Позволяет обходить поддиапазоны полного диапазона iterindex [0, NpyIter_IterSize(iter)). Нужный диапазон задаётся функцией NpyIter_ResetToIterIndexRange.

Этот флаг можно использовать с NPY_ITER_EXTERNAL_LOOP только при включённом NPY_ITER_BUFFERED. Без буферизации размер внутреннего цикла всегда равен размеру самого внутреннего измерения обхода. Разбиение такого цикла потребовало бы специальной обработки, фактически приближая реализацию к буферизованной.

NPY_ITER_BUFFERED#

Заставляет итератор хранить данные для буферизации и использовать буферы для выполнения требований к типу данных, выравниванию и порядку байтов. Чтобы буферизовать операнд, не задавайте флаги NPY_ITER_COPY и NPY_ITER_UPDATEIFCOPY: они имеют приоритет над буферизацией. Буферизация особенно полезна при использовании итератора из кода Python: обработка более крупных блоков данных за один раз позволяет уменьшить относительные накладные расходы интерпретатора.

При использовании с NPY_ITER_EXTERNAL_LOOP внутренний цикл вызывающего кода может получать более крупные блоки, чем без буферизации, благодаря расположению шагов в памяти.

Если операнду задан флаг NPY_ITER_COPY или NPY_ITER_UPDATEIFCOPY, предпочтение отдаётся созданию копии, а не буферизации. Буферизация всё равно применяется после трансляции массива, если для получения постоянного шага нужно дублировать элементы.

При обычной буферизации длина каждого внутреннего цикла равна размеру буфера либо может быть больше при NPY_ITER_GROWINNER. Если включён NPY_ITER_REDUCE_OK и выполняется свёртка, длина внутренних циклов может уменьшаться в зависимости от структуры свёртки.

NPY_ITER_GROWINNER#

При включённой буферизации позволяет увеличивать длину внутреннего цикла, когда буферизация не нужна. Этот параметр лучше подходит для однократного прохода по всем данным, чем для обработки небольших, удобных для кеша массивов временных значений на каждом внутреннем цикле.

NPY_ITER_DELAY_BUFALLOC#

Когда буферизация включена, выделение буферов задерживается до тех пор, пока не будет вызвана NpyIter_Reset или другая функция сброса. Этот флаг существует, чтобы избежать ненужного копирования данных буфера при создании нескольких копий буферизованного итератора для многопоточной итерации.

Этот флаг также используется для настройки операций свёртки. После создания итератора и автоматического выделения им выходного массива свёртки (обязательно с доступом READWRITE) этот массив можно инициализировать нейтральным элементом операции. Чтобы получить объект, используйте NpyIter_GetOperandArray. Затем вызовите NpyIter_Reset для выделения буферов и заполнения их начальными значениями.

NPY_ITER_COPY_IF_OVERLAP#

Если какой-либо операнд записи перекрывается с каким-либо операндом чтения, устраните все перекрытия, создав временные копии (при необходимости включив UPDATEIFCOPY для операндов записи). Пара операндов перекрывается, если существует адрес памяти, содержащий данные, общие для обоих массивов.

Точное обнаружение перекрытия требует времени, растущего экспоненциально с числом измерений. Поэтому используется эвристика: возможны ложноположительные результаты, то есть лишние копии в необычных случаях, но ложноотрицательных результатов нет.

Если какое-либо перекрытие чтения/записи существует, этот флаг гарантирует, что результат операции такой же, как если бы все операнды были скопированы.В тех случаях, когда копии должны быть сделаны, результат вычисления может быть неопределенным без этого флага!

Флаги, которые можно передать в op_flags[i], где 0 <= i < nop:

NPY_ITER_READWRITE#
NPY_ITER_READONLY#
NPY_ITER_WRITEONLY#

Определяют, как пользователь итератора будет читать op[i] или записывать в него. Для каждого операнда необходимо указать ровно один из этих флагов. Флаги NPY_ITER_READWRITE и NPY_ITER_WRITEONLY для операнда, предоставленного пользователем, могут включить поведение WRITEBACKIFCOPY. При вызове NpyIter_Deallocate данные будут записаны обратно в исходный массив.

NPY_ITER_COPY#

Разрешает создание копии op[i], если исходный объект не соответствует требованиям к типу данных или выравниванию, заданным флагами и параметрами конструктора.

NPY_ITER_UPDATEIFCOPY#

Включает NPY_ITER_COPY. Если операнд-массив помечен для записи и скопирован, данные копии записываются обратно в op[i] при вызове NpyIter_Deallocate.

Если операнд помечен как доступный только для записи и требуется копия, создаётся неинициализированный временный массив. Его содержимое копируется обратно в op[i] при вызове NpyIter_Deallocate; ненужное исходное копирование при этом не выполняется.

NPY_ITER_NBO#
NPY_ITER_ALIGNED#
NPY_ITER_CONTIG#

Предписывает итератору предоставлять для op[i] данные с родным порядком байтов, выравниванием согласно требованиям dtype, непрерывным расположением в памяти или любой комбинацией этих свойств.

По умолчанию итератор создает указатели на предоставленные массивы, которые могут быть выровнены или невыровнены и иметь любой порядок байтов. Если копирование или буферизация не включены и данные операнда не удовлетворяют ограничениям, будет выдана ошибка.

Требование непрерывного расположения в памяти относится только к внутреннему циклу; между последовательными внутренними циклами указатели могут изменяться произвольно.

Если запрошенный тип данных имеет несистемный порядок байтов, флаг NBO переопределяет его и преобразует тип к системному порядку.

NPY_ITER_ALLOCATE#

Предназначено для выходных массивов; требует флага NPY_ITER_WRITEONLY или NPY_ITER_READWRITE. Если op[i] равен NULL, создаёт массив с окончательной формой после трансляции и расположением данных, соответствующим порядку обхода итератора.

Если op[i] равен NULL, запрашиваемый тип данных op_dtypes[i] тоже может быть NULL. Тогда он автоматически определяется по dtype массивов, помеченных для чтения, по тем же правилам, что в универсальных функциях. Важна обработка порядка байтов: при ровно одном входе его dtype используется без изменений; при объединении нескольких входных dtype результат имеет машинный порядок байтов.

После выделения массива с этим флагом вызывающий код может получить новый массив, вызвав NpyIter_GetOperandArray и взяв i-й объект из возвращённого массива C. Чтобы получить собственную ссылку на массив, необходимо вызвать для него Py_INCREF.

NPY_ITER_NO_SUBTYPE#

При использовании с NPY_ITER_ALLOCATE этот флаг запрещает создавать выходной массив подкласса и принудительно создаёт обычный ndarray.

NPY_ITER_NO_BROADCAST#

Обеспечивает точное совпадение размеров входного или выходного массива с измерениями обхода.

NPY_ITER_ARRAYMASK#

Указывается, что этот операнд является маской, используемой для выбора элементов при записи в операнды, на которых к ним приложен флаг NPY_ITER_WRITEMASKED. Только один операнд может иметь флаг NPY_ITER_ARRAYMASK.

Тип данных операнда с этим флагом должен быть NPY_BOOL, NPY_MASK или структурированным dtype, все поля которого имеют допустимые для маски dtype. В последнем случае структура должна соответствовать структурированному операнду с флагом WRITEMASKED, поскольку маска задаётся отдельно для каждого его поля.

Этот флаг влияет только на обратную запись из буфера в массив. Поэтому, если операнд также имеет флаг NPY_ITER_READWRITE или NPY_ITER_WRITEONLY, код обхода может записывать значения в этот операнд, управляя тем, какие элементы останутся неизменными, а какие будут изменены. Это полезно, когда маска должна объединять входные маски.

NPY_ITER_WRITEMASKED#

Этот массив служит маской для всех операндов writemasked. Флаг writemasked означает, что запись выполняется только для элементов, которым соответствует True в выбранном операнде ARRAYMASK. Обычно итератор не обеспечивает этого сам: соблюдать условие должен код обхода.

Если используется флаг writemasked и этот операнд буферизуется, способ копирования данных из буфера в массив меняется. Используется копирование с маской: копируются лишь те элементы буфера, для соответствующих элементов операнда ARRAYMASK которых writemasked возвращает истинное значение.

NPY_ITER_OVERLAP_ASSUME_ELEMENTWISE#

При проверке перекрытия памяти предположите, что доступ к операндам с включенным NPY_ITER_OVERLAP_ASSUME_ELEMENTWISE осуществляется только в порядке итератора.

Это позволяет итератору рассуждать о зависимости данных, возможно, избегая ненужных копий.

Этот флаг действует только в том случае, если на итераторе включен NPY_ITER_COPY_IF_OVERLAP.

NpyIter *NpyIter_AdvancedNew(npy_intp nop, PyArrayObject **op, npy_uint32 flags, NPY_ORDER order, NPY_CASTING casting, npy_uint32 *op_flags, PyArray_Descr **op_dtypes, int oa_ndim, int **op_axes, npy_intp const *itershape, npy_intp buffersize)#

Расширяет NpyIter_MultiNew несколькими расширенными опциями, обеспечивающими больший контроль над трансляцией массивов и буферизацией.

Если значения -1/NULL передаются в oa_ndim, op_axes, itershape и buffersize, это эквивалентно NpyIter_MultiNew.

Если параметр oa_ndim не равен 0 или -1, он задаёт число измерений, обход которых выполняется с настраиваемой трансляцией массивов. При его задании обязательно указать op_axes; также можно указать itershape. Параметр op_axes позволяет подробно настроить сопоставление и обход осей массивов-операндов. В op_axes нужно передать массив из nop указателей на массивы длины oa_ndim типа npy_intp. Если элемент op_axes равен NULL, применяются обычные правила трансляции массивов. В op_axes[j][i] хранится либо допустимая ось op[j], либо -1, что означает newaxis. В каждом массиве op_axes[j] оси не должны повторяться. Следующий пример показывает обычную трансляцию трёхмерного, двумерного, одномерного массивов и скаляра.

Примечание: до NumPy 1.8 значение oa_ndim == 0 означало, что op_axes и itershape не используются. Это устаревшее поведение: вместо 0 нужно указывать -1. Для лучшей обратной совместимости в таком случае можно использовать NpyIter_MultiNew.

int oa_ndim = 3;               /* # iteration axes */
int op0_axes[] = {0, 1, 2};    /* 3-D operand */
int op1_axes[] = {-1, 0, 1};   /* 2-D operand */
int op2_axes[] = {-1, -1, 0};  /* 1-D operand */
int op3_axes[] = {-1, -1, -1}  /* 0-D (scalar) operand */
int* op_axes[] = {op0_axes, op1_axes, op2_axes, op3_axes};

  • Одномерный операнд
  • Нульмерный (скалярный) операнд

Параметр itershape задаёт конкретную форму обхода итератора. Это массив длины oa_ndim; отрицательные элементы означают, что размер определяется по операндам. Параметр позволяет автоматически выделяемым выходным массивам иметь дополнительные измерения, не соответствующие ни одному измерению входов.

Если buffersize равен нулю, используется размер буфера по умолчанию; иначе он задаёт размер используемого буфера. Рекомендуются размеры, равные степеням двойки, например 4096 или 8192.

Возвращает NULL в случае ошибки, в противном случае возвращает выделенный итератор.

Для многопоточного обхода рекомендуется сначала создать итератор с флагами NPY_ITER_EXTERNAL_LOOP, NPY_ITER_RANGED, NPY_ITER_BUFFERED, NPY_ITER_DELAY_BUFALLOC и, возможно, NPY_ITER_GROWINNER. Функции Reset* и некоторые другие можно безопасно вызывать, передав ненулевой указатель в параметре errmsg: тогда они возвращают ошибки через него, а не устанавливают исключение Python.

int NpyIter_RemoveAxis(NpyIter *iter, int axis)#

Исключает ось из обхода. При создании итератора должен быть установлен флаг NPY_ITER_MULTI_INDEX. Функция не работает при включённой буферизации или отслеживании индекса. Она также возвращает итератор в исходное состояние.

Это полезно, например, для организации цикла накопления. Сначала можно создать итератор со всеми измерениями, включая ось накопления, чтобы правильно создать выходной массив. Затем ось накопления исключают из итератора и выполняют вычисление с помощью вложенного цикла.

Создание итераторов для вложенного обхода требует аккуратности. Операнды всех итераторов должны полностью совпадать, иначе вызовы NpyIter_ResetBasePointers будут некорректны. Поэтому автоматическое копирование и выделение выходных массивов нельзя применять без учёта этого требования. При этом можно пользоваться автоматическим преобразованием данных и приведением типов: создайте один итератор со всеми нужными параметрами преобразования, получите выделенные операнды с помощью NpyIter_GetOperandArray и передайте их в конструкторы остальных итераторов.

ПРЕДУПРЕЖДЕНИЕ: при создании итераторов для вложенного обхода одно и то же измерение нельзя использовать в нескольких итераторах. Иначе при обходе будут возникать указатели за пределами массива.

ПРЕДУПРЕЖДЕНИЕ: для вложенного обхода буферизацию можно включать только у самого внутреннего итератора. Если использовать буферизованный итератор как источник baseptrs, указатели будут направлены в небольшой буфер, а не в массив, и внутренний обход будет некорректным.

Модель использования вложенных итераторов выглядит следующим образом.

NpyIter *iter1, *iter1;
NpyIter_IterNextFunc *iternext1, *iternext2;
char **dataptrs1;

/*
 * With the exact same operands, no copies allowed, and
 * no axis in op_axes used both in iter1 and iter2.
 * Buffering may be enabled for iter2, but not for iter1.
 */
iter1 = ...; iter2 = ...;

iternext1 = NpyIter_GetIterNext(iter1);
iternext2 = NpyIter_GetIterNext(iter2);
dataptrs1 = NpyIter_GetDataPtrArray(iter1);

do {
    NpyIter_ResetBasePointers(iter2, dataptrs1);
    do {
        /* Use the iter2 values */
    } while (iternext2(iter2));
} while (iternext1(iter1));
Комментарии к примеру на русском
  • С теми же операндами, без разрешения копирования; ни одна ось op_axes не используется одновременно в iter1 и iter2. Для iter2 буферизация допустима, для iter1 — нет.
  • Используем значения iter2

int NpyIter_CreateCompatibleStrides(NpyIter *iter, npy_intp itemsize, npy_intp *outstrides)#

Строит набор шагов, совпадающих с шагами выходного массива, созданного с флагом NPY_ITER_ALLOCATE при op_axes=NULL. Это соответствует данным, непрерывно расположенным в памяти, но необязательно в порядке C или Fortran. Следует использовать вместе с NpyIter_GetShape и NpyIter_GetNDim, передав конструктору флаг NPY_ITER_MULTI_INDEX.

Одно из применений функции — повторить форму и расположение данных итератора, добавив одно или несколько измерений. Например, чтобы получить вектор градиента для каждого входного значения, передайте ndim*itemsize как itemsize, затем добавьте в конце измерение размера ndim с шагом itemsize. Для матрицы Гессе аналогично добавляются два измерения; можно также использовать симметрию и упаковать значения в одно измерение по выбранной схеме.

Функцию можно вызывать только при отслеживании мультииндекса итератором и использовании NPY_ITER_DONT_NEGATE_STRIDES для запрета обратного обхода осей.

Для массива, созданного этим методом, прибавление itemsize на каждой итерации обеспечивает обход в порядке, соответствующем итератору.

Возвращает NPY_SUCCEED или NPY_FAIL.

npy_bool NpyIter_IsFirstVisit(NpyIter *iter, int iop)#

Проверяет, впервые ли итератор обращается к текущим элементам указанного операнда свёртки. Ответ корректен для операндов свёртки, а также при отключённой буферизации. Для буферизованных операндов, не участвующих в свёртке, ответ может быть неверным.

Эта функция предназначена для использования только в режиме EXTERNAL_LOOP и будет давать неправильные ответы, когда этот режим не включен.

Если функция возвращает истину, вызывающий код должен также проверить шаг операнда во внутреннем цикле. При нулевом шаге только первый элемент самого внутреннего внешнего цикла посещается впервые.

ПРЕДУПРЕЖДЕНИЕ: ради производительности не проверяется, находится ли ‘iop’ в допустимых границах, является ли он операндом свёртки и включён ли режим EXTERNAL_LOOP. Эти проверки должен выполнять вызывающий код, причём за пределами внутренних циклов.

Функции для итерации#

NpyIter_IterNextFunc *NpyIter_GetIterNext(NpyIter *iter, char **errmsg)#

Возвращает указатель на функцию обхода. Специализированный вариант функции может определяться при этом вызове, а не храниться в структуре итератора. Поэтому для высокой производительности указатель нужно сохранить в переменной, а не получать заново на каждой итерации цикла.

Возвращает NULL при ошибке. Если errmsg не равен NULL, при возврате NPY_FAIL исключение Python не устанавливается; вместо этого в *errmsg записывается сообщение об ошибке. При ненулевом errmsg функцию можно безопасно вызывать без удержания GIL Python.

Типичная конструкция цикла выглядит следующим образом.

NpyIter_IterNextFunc *iternext = NpyIter_GetIterNext(iter, NULL);
char** dataptr = NpyIter_GetDataPtrArray(iter);

do {
    /* use the addresses dataptr[0], ... dataptr[nop-1] */
} while(iternext(iter));
Комментарии к примеру на русском
  • Используем адреса dataptr[0], …, dataptr[nop-1]

При указании NPY_ITER_EXTERNAL_LOOP типичная конструкция внутреннего цикла выглядит так.

NpyIter_IterNextFunc *iternext = NpyIter_GetIterNext(iter, NULL);
char** dataptr = NpyIter_GetDataPtrArray(iter);
npy_intp* stride = NpyIter_GetInnerStrideArray(iter);
npy_intp* size_ptr = NpyIter_GetInnerLoopSizePtr(iter), size;
npy_intp iop, nop = NpyIter_GetNOp(iter);

do {
    size = *size_ptr;
    while (size--) {
        /* use the addresses dataptr[0], ... dataptr[nop-1] */
        for (iop = 0; iop < nop; ++iop) {
            dataptr[iop] += stride[iop];
        }
    }
} while (iternext());
Комментарии к примеру на русском
  • Используем адреса dataptr[0], …, dataptr[nop-1]

Обратите внимание: мы используем массив dataptr внутри итератора, не копируя значения во временную локальную переменную. Это возможно, поскольку при вызове iternext() указатели заменяются новыми значениями, а не обновляются приращением.

Если используется буфер фиксированного при компиляции размера (оба флага NPY_ITER_BUFFERED и NPY_ITER_EXTERNAL_LOOP), размер внутреннего цикла можно также использовать как сигнал завершения. Гарантируется, что размер станет нулевым, когда iternext() вернёт false, что позволяет построить следующий цикл. При таком подходе не передавайте флаг NPY_ITER_GROWINNER: в некоторых условиях он увеличивает размер.

/* The constructor should have buffersize passed as this value */
#define FIXED_BUFFER_SIZE 1024

NpyIter_IterNextFunc *iternext = NpyIter_GetIterNext(iter, NULL);
char **dataptr = NpyIter_GetDataPtrArray(iter);
npy_intp *stride = NpyIter_GetInnerStrideArray(iter);
npy_intp *size_ptr = NpyIter_GetInnerLoopSizePtr(iter), size;
npy_intp i, iop, nop = NpyIter_GetNOp(iter);

/* One loop with a fixed inner size */
size = *size_ptr;
while (size == FIXED_BUFFER_SIZE) {
    /*
     * This loop could be manually unrolled by a factor
     * which divides into FIXED_BUFFER_SIZE
     */
    for (i = 0; i < FIXED_BUFFER_SIZE; ++i) {
        /* use the addresses dataptr[0], ... dataptr[nop-1] */
        for (iop = 0; iop < nop; ++iop) {
            dataptr[iop] += stride[iop];
        }
    }
    iternext();
    size = *size_ptr;
}

/* Finish-up loop with variable inner size */
if (size > 0) do {
    size = *size_ptr;
    while (size--) {
        /* use the addresses dataptr[0], ... dataptr[nop-1] */
        for (iop = 0; iop < nop; ++iop) {
            dataptr[iop] += stride[iop];
        }
    }
} while (iternext());
Комментарии к примеру на русском
  • Это значение нужно передать конструктору как buffersize
  • Один цикл с фиксированным внутренним размером
  • Этот цикл можно вручную развернуть с коэффициентом, на который делится FIXED_BUFFER_SIZE
  • Используем адреса dataptr[0], …, dataptr[nop-1]
  • Завершающий цикл с переменным внутренним размером

Переход с прежних итераторов NumPy#

Старый API итератора включает функции PyArrayIter_Check, PyArray_Iter* и PyArray_ITER_*. API мультиитератора включает PyArray_MultiIter*, PyArray_Broadcast и PyArray_RemoveSmallest.
