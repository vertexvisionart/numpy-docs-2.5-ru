Создание собственной ufunc#

У меня есть Сила!
— He-Man

Создание новой универсальной функции #

Перед чтением полезно познакомиться с основами расширений Python на C: просмотрите руководства в разделе 1 Расширение и встраивание интерпретатора Python и в разделе Как расширить NumPy.

Модуль umath — автоматически генерируемый модуль на C, создающий множество универсальных функций (ufunc). Он содержит много примеров их создания. Написать собственную универсальную функцию, использующую этот механизм, тоже несложно. Предположим, у вас есть функция, которую требуется применять поэлементно к входным данным. Создав для неё ufunc, вы получите функцию, которая поддерживает

  • Трансляция массивов

  • N-мерный обход

  • Автоматические преобразования типов с минимальным использованием памяти

  • необязательные выходные массивы

Создать собственную ufunc несложно: для каждого поддерживаемого типа данных нужен одномерный цикл с определённой сигнатурой. Здесь используются только типы с фиксированным размером элемента. Ниже приведена функция создания ufunc для встроенных типов данных; для регистрации ufunc для пользовательских типов применяется другой механизм.

В следующих разделах приведены примеры кода, которые легко изменить для создания собственных универсальных функций. Это всё более полные или сложные варианты функции logit (логит), часто используемой в статистическом моделировании. Она интересна ещё и тем, что благодаря стандартам IEEE (в частности, IEEE 754) все создаваемые ниже варианты logit автоматически обладают следующим поведением.

>>> logit(0)
-inf
>>> logit(1)
inf
>>> logit(2)
nan
>>> logit(-2)
nan

Это удобно: автору функции не нужно вручную обеспечивать распространение inf и nan.

Пример расширения без ufunc#

Для сравнения и лучшего понимания приведём простую реализацию расширения на C для logit без использования NumPy.

Для этого нам нужно три файла. Первый - это файл C, который содержит фактический код, а остальные - два файла проекта, которые описывают, как создать модуль.

#define PY_SSIZE_T_CLEAN
#include <Python.h>
#include <math.h>

/*
 * spammodule.c
 * This is the C code for a non-numpy Python extension to
 * define the logit function, where logit(p) = log(p/(1-p)).
 * This function will not work on numpy arrays automatically.
 * numpy.vectorize must be called in python to generate
 * a numpy-friendly function.
 *
 * Details explaining the Python-C API can be found under
 * 'Extending and Embedding' and 'Python/C API' at
 * docs.python.org .
 */


/* This declares the logit function */
static PyObject *spam_logit(PyObject *self, PyObject *args);

/*
 * This tells Python what methods this module has.
 * See the Python-C API for more information.
 */
static PyMethodDef SpamMethods[] = {
    {"logit",
        spam_logit,
        METH_VARARGS, "compute logit"},
    {NULL, NULL, 0, NULL}
};

/*
 * This actually defines the logit function for
 * input args from Python.
 */

static PyObject *spam_logit(PyObject *self, PyObject *args)
{
    double p;

    /* This parses the Python argument into a double */
    if(!PyArg_ParseTuple(args, "d", &p)) {
        return NULL;
    }

    /* THE ACTUAL LOGIT FUNCTION */
    p = p/(1-p);
    p = log(p);

    /*This builds the answer back into a python object */
    return Py_BuildValue("d", p);
}

/* This initiates the module using the above definitions. */
static struct PyModuleDef moduledef = {
    PyModuleDef_HEAD_INIT,
    "spam",
    NULL,
    -1,
    SpamMethods,
    NULL,
    NULL,
    NULL,
    NULL
};

PyMODINIT_FUNC PyInit_spam(void)
{
    PyObject *m;
    m = PyModule_Create(&moduledef);
    if (!m) {
        return NULL;
    }
    return m;
}
Комментарии к примеру на русском
  • spammodule.c. Код C для расширения Python без NumPy, определяющего функцию logit(p) = log(p/(1-p)). Она не работает с массивами NumPy автоматически. Для получения совместимой функции нужно вызвать numpy.vectorize в Python. Подробности Python C API см. в разделах «Extending and Embedding» и «Python/C API» на docs.python.org.
  • Объявляет функцию logit
  • Сообщает Python, какие методы есть в модуле. Подробнее см. Python C API.
  • Определяет функцию logit, принимающую аргументы из Python.
  • Разбирает аргумент Python и преобразует его в double
  • САМО ВЫЧИСЛЕНИЕ LOGIT
  • Преобразует ответ обратно в объект Python
  • Инициализирует модуль с использованием определений выше.

Модуль создаётся так же, как пакет Python: сначала создают файл pyproject.toml с указанием системы сборки, затем ещё один файл для этой системы, описывающий компиляцию кода. Мы рекомендуем meson-python, поскольку используем эту систему для самого NumPy, но ниже также показано использование более старой системы setuptools.

Примеры pyproject.toml и meson.build.

[project]

version = "0.1"

[build-system]
requires = ["meson-python"]
build-backend = "mesonpy"

project('spam', 'c')

py = import('python').find_installation()

sources = files('spammodule.c')

extension_module = py.extension_module(
  'spam',
  sources,
  install: true,
)

Примеры pyproject.toml и setup.py.

version = "0.1"

[build-system]

build-backend = "setuptools.build_meta"

setup(name='spam', version='1.0',
      ext_modules=[spammodule])

Используя любой из вышеперечисленных способов, можно собрать и установить пакет spam с помощью:

pip install .

После импорта модуля spam в Python функцию logit можно вызвать как spam.logit. Показанная выше функция сама по себе не работает с массивами NumPy: для этого к ней нужно применить numpy.vectorize. Например:

>>> import numpy as np
>>> import spam
>>> spam.logit(0)
-inf
>>> spam.logit(1)
inf
>>> spam.logit(0.5)
0.0
>>> x = np.linspace(0,1,10)
>>> spam.logit(x)
TypeError: only length-1 arrays can be converted to Python scalars
>>> f = np.vectorize(spam.logit)
>>> f(x)
array([       -inf, -2.07944154, -1.25276297, -0.69314718, -0.22314355,
    0.22314355,  0.69314718,  1.25276297,  2.07944154,         inf])

ПОЛУЧЕННАЯ ФУНКЦИЯ LOGIT НЕ БУДЕТ БЫСТРОЙ! numpy.vectorize просто вызывает spam.logit в цикле. Хотя цикл выполняется на уровне C, массив NumPy постоянно разбирается и создаётся заново, что требует значительных затрат. При сравнении numpy.vectorize(spam.logit) с универсальными функциями logit, создаваемыми ниже, автор получил почти четырёхкратное ускорение для универсальных функций. В зависимости от характера функции ускорение, конечно, может быть больше или меньше.

Пример универсальной функции NumPy для одного dtype#

Для простоты создадим универсальную функцию для одного типа данных: 'f8', то есть double. Как и в предыдущем разделе, сначала приведём файл .c, затем файлы для создания модуля npufunc с этой универсальной функцией.

Участок кода, выполняющий вычисления универсальной функции (ufunc), отмечен строками /* BEGIN main ufunc computation */ и /* END main ufunc computation */. Чтобы создать собственную универсальную функцию, прежде всего нужно изменить код между этими строками.

#define PY_SSIZE_T_CLEAN
#include <Python.h>
#include "numpy/ndarraytypes.h"
#include "numpy/ufuncobject.h"
#include "numpy/npy_3kcompat.h"
#include <math.h>

/*
 * single_type_logit.c
 * This is the C code for creating your own
 * NumPy ufunc for a logit function.
 *
 * In this code we only define the ufunc for
 * a single dtype. The computations that must
 * be replaced to create a ufunc for
 * a different function are marked with BEGIN
 * and END.
 *
 * Details explaining the Python-C API can be found under
 * 'Extending and Embedding' and 'Python/C API' at
 * docs.python.org .
 */

static PyMethodDef LogitMethods[] = {
    {NULL, NULL, 0, NULL}
};

/* The loop definition must precede the PyMODINIT_FUNC. */

static void double_logit(char **args, const npy_intp *dimensions,
                         const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp n = dimensions[0];
    char *in = args[0], *out = args[1];
    npy_intp in_step = steps[0], out_step = steps[1];

    double tmp;

    for (i = 0; i < n; i++) {
        /* BEGIN main ufunc computation */
        tmp = *(double *)in;
        tmp /= 1 - tmp;
        *((double *)out) = log(tmp);
        /* END main ufunc computation */

        in += in_step;
        out += out_step;
    }
}

/* This a pointer to the above function */
PyUFuncGenericFunction funcs[1] = {&double_logit};

/* These are the input and return dtypes of logit.*/
static const char types[2] = {NPY_DOUBLE, NPY_DOUBLE};

static struct PyModuleDef moduledef = {
    PyModuleDef_HEAD_INIT,
    "npufunc",
    NULL,
    -1,
    LogitMethods,
    NULL,
    NULL,
    NULL,
    NULL
};

PyMODINIT_FUNC PyInit_npufunc(void)
{
    PyObject *m, *logit, *d;

    import_array();
    import_umath();

    m = PyModule_Create(&moduledef);
    if (!m) {
        return NULL;
    }

    logit = PyUFunc_FromFuncAndData(funcs, NULL, types, 1, 1, 1,
                                    PyUFunc_None, "logit",
                                    "logit_docstring", 0);

    d = PyModule_GetDict(m);

    PyDict_SetItemString(d, "logit", logit);
    Py_DECREF(logit);

    return m;
}

Вычисления, которые нужно заменить для реализации другой функции, отмечены BEGIN и END. Подробности Python C API см. в разделах «Extending and Embedding» и «Python/C API» на docs.python.org.
  • Определение цикла должно находиться перед PyMODINIT_FUNC.
  • BEGIN: основное вычисление ufunc
  • END: основное вычисление ufunc
  • Указатель на функцию выше
  • Входной и выходной dtype функции logit.

Для файлов, необходимых для создания модуля, главное отличие от нашего предыдущего примера заключается в том, что теперь нам нужно объявить зависимости от numpy.

Пример универсальной функции NumPy с несколькими dtype#

Теперь расширим пример до полноценной универсальной функции logit с внутренними циклами для float, double и long double. Можно использовать те же файлы сборки, заменив исходный файл single_type_logit.c на multi_type_logit.c.

Места в коде, соответствующие фактическим вычислениям для ufunc, отмечены /* BEGIN main ufunc computation */ и /* END main ufunc computation */. Код между этими строками является основным, что необходимо изменить, чтобы создать свой собственный ufunc.

#define PY_SSIZE_T_CLEAN
#include <Python.h>
#include "numpy/ndarraytypes.h"
#include "numpy/ufuncobject.h"
#include <math.h>

/*
 * multi_type_logit.c
 * This is the C code for creating your own
 * NumPy ufunc for a logit function.
 *
 * Each function of the form type_logit defines the
 * logit function for a different numpy dtype. Each
 * of these functions must be modified when you
 * create your own ufunc. The computations that must
 * be replaced to create a ufunc for
 * a different function are marked with BEGIN
 * and END.
 *
 * Details explaining the Python-C API can be found under
 * 'Extending and Embedding' and 'Python/C API' at
 * docs.python.org .
 *
 */

static PyMethodDef LogitMethods[] = {
    {NULL, NULL, 0, NULL}
};

/* The loop definitions must precede the PyMODINIT_FUNC. */

static void long_double_logit(char **args, const npy_intp *dimensions,
                              const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp n = dimensions[0];
    char *in = args[0], *out = args[1];
    npy_intp in_step = steps[0], out_step = steps[1];

    long double tmp;

    for (i = 0; i < n; i++) {
        /* BEGIN main ufunc computation */
        tmp = *(long double *)in;
        tmp /= 1 - tmp;
        *((long double *)out) = logl(tmp);
        /* END main ufunc computation */

        in += in_step;
        out += out_step;
    }
}

static void double_logit(char **args, const npy_intp *dimensions,
                         const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp n = dimensions[0];
    char *in = args[0], *out = args[1];
    npy_intp in_step = steps[0], out_step = steps[1];

    double tmp;

    for (i = 0; i < n; i++) {
        /* BEGIN main ufunc computation */
        tmp = *(double *)in;
        tmp /= 1 - tmp;
        *((double *)out) = log(tmp);
        /* END main ufunc computation */

        in += in_step;
        out += out_step;
    }
}

static void float_logit(char **args, const npy_intp *dimensions,
                       const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp n = dimensions[0];
    char *in = args[0], *out = args[1];
    npy_intp in_step = steps[0], out_step = steps[1];

    float tmp;

    for (i = 0; i < n; i++) {
        /* BEGIN main ufunc computation */
        tmp = *(float *)in;
        tmp /= 1 - tmp;
        *((float *)out) = logf(tmp);
        /* END main ufunc computation */

        in += in_step;
        out += out_step;
    }
}



/*This gives pointers to the above functions*/
PyUFuncGenericFunction funcs[3] = {&float_logit,
                                   &double_logit,
                                   &long_double_logit};

static const char types[6] = {NPY_FLOAT, NPY_FLOAT,
                              NPY_DOUBLE, NPY_DOUBLE,
                              NPY_LONGDOUBLE, NPY_LONGDOUBLE};

static struct PyModuleDef moduledef = {
    PyModuleDef_HEAD_INIT,
    "npufunc",
    NULL,
    -1,
    LogitMethods,
    NULL,
    NULL,
    NULL,
    NULL
};

PyMODINIT_FUNC PyInit_npufunc(void)
{
    PyObject *m, *logit, *d;

    import_array();
    import_umath();

    m = PyModule_Create(&moduledef);
    if (!m) {
        return NULL;
    }

    logit = PyUFunc_FromFuncAndData(funcs, NULL, types, 4, 1, 1,
                                    PyUFunc_None, "logit",
                                    "logit_docstring", 0);

    d = PyModule_GetDict(m);

    PyDict_SetItemString(d, "logit", logit);
    Py_DECREF(logit);

    return m;
}

При создании своей ufunc нужно изменить каждую из них. Вычисления для замены отмечены BEGIN и END. Подробности Python C API см. в разделах «Extending and Embedding» и «Python/C API» на docs.python.org.

>>> import numpy as np
>>> import npufunc
>>> npufunc.logit(0.5)

>>> a = np.linspace(0, 1, 5, dtype=np.float32)
>>> npufunc.logit(a)
<python-input-4>:1: RuntimeWarning: divide by zero encountered in logit
array([      -inf, -1.0986123,  0.       ,  1.0986123,        inf],

Примечание

Поддержка float16, то есть половинной точности, в собственной ufunc сложнее из-за нестандартного представления в C и необходимых преобразований. Приведённый выше код принимает вход float16, но преобразует его в float32. Результат также имеет тип float32, однако его можно преобразовать обратно в float16, передав подходящий выходной массив, например npufunc.logit(a, out=np.empty_like(a)). Примеры настоящих циклов float16 см. в исходном коде NumPy.

Пример ufunc NumPy с несколькими аргументами и возвращаемыми значениями#

Создать универсальную функцию с несколькими аргументами несложно. Изменим код универсальной функции logit так, чтобы вычислять (A * B, logit(A * B)). Для простоты создадим внутренний цикл только для double.

Мы снова указываем только код C, поскольку файлы, необходимые для создания модуля, такие же, как и раньше, но имя исходного файла заменено на multi_arg_logit.c.

Ниже приведён файл C. Создаваемая универсальная функция принимает аргументы A и B и возвращает кортеж с первым элементом A * B и вторым logit(A * B). Она автоматически поддерживает трансляцию массивов (согласование форм) и остальные свойства универсальных функций.

#define PY_SSIZE_T_CLEAN
#include <Python.h>
#include "numpy/ndarraytypes.h"
#include "numpy/ufuncobject.h"
#include <math.h>

/*
 * multi_arg_logit.c
 * This is the C code for creating your own
 * NumPy ufunc for a multiple argument, multiple
 * return value ufunc. The places where the
 * ufunc computation is carried out are marked
 * with comments.
 *
 * Details explaining the Python-C API can be found under
 * 'Extending and Embedding' and 'Python/C API' at
 * docs.python.org.
 */

static PyMethodDef LogitMethods[] = {
    {NULL, NULL, 0, NULL}
};

/* The loop definition must precede the PyMODINIT_FUNC. */

static void double_logitprod(char **args, const npy_intp *dimensions,
                             const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp n = dimensions[0];
    char *in1 = args[0], *in2 = args[1];
    char *out1 = args[2], *out2 = args[3];
    npy_intp in1_step = steps[0], in2_step = steps[1];
    npy_intp out1_step = steps[2], out2_step = steps[3];

    double tmp;

    for (i = 0; i < n; i++) {
        /* BEGIN main ufunc computation */
        tmp = *(double *)in1;
        tmp *= *(double *)in2;
        *((double *)out1) = tmp;
        *((double *)out2) = log(tmp / (1 - tmp));
        /* END main ufunc computation */

        in1 += in1_step;
        in2 += in2_step;
        out1 += out1_step;
        out2 += out2_step;
    }
}

/*This a pointer to the above function*/
PyUFuncGenericFunction funcs[1] = {&double_logitprod};

/* These are the input and return dtypes of logit.*/

static const char types[4] = {NPY_DOUBLE, NPY_DOUBLE,
                              NPY_DOUBLE, NPY_DOUBLE};

static struct PyModuleDef moduledef = {
    PyModuleDef_HEAD_INIT,
    "npufunc",
    NULL,
    -1,
    LogitMethods,
    NULL,
    NULL,
    NULL,
    NULL
};

PyMODINIT_FUNC PyInit_npufunc(void)
{
    PyObject *m, *logit, *d;

    import_array();
    import_umath();

    m = PyModule_Create(&moduledef);
    if (!m) {
        return NULL;
    }

    logit = PyUFunc_FromFuncAndData(funcs, NULL, types, 1, 2, 2,
                                    PyUFunc_None, "logit",
                                    "logit_docstring", 0);

    d = PyModule_GetDict(m);

    PyDict_SetItemString(d, "logit", logit);
    Py_DECREF(logit);

    return m;
}

Пример универсальной функции NumPy с аргументами структурированного типа данных#

Этот пример показывает создание универсальной функции для структурированного типа данных массива. Простая функция складывает два массива с dtype 'u8,u8,u8'. Порядок действий немного отличается от остальных примеров: вызов PyUFunc_FromFuncAndData не регистрирует циклы универсальной функции для пользовательских и структурированных типов данных. Для завершения настройки также нужно вызвать PyUFunc_RegisterLoopForDescr.

Мы приводим только код C: файлы, необходимые для построения модуля, точно такие же, как раньше, за исключением того, что исходный файл теперь называется add_triplet.c.

#define PY_SSIZE_T_CLEAN
#include <Python.h>
#include "numpy/ndarraytypes.h"
#include "numpy/ufuncobject.h"
#include "numpy/npy_3kcompat.h"
#include <math.h>

/*
 * add_triplet.c
 * This is the C code for creating your own
 * NumPy ufunc for a structured array dtype.
 *
 * Details explaining the Python-C API can be found under
 * 'Extending and Embedding' and 'Python/C API' at
 * docs.python.org.
 */

static PyMethodDef StructUfuncTestMethods[] = {
    {NULL, NULL, 0, NULL}
};

/* The loop definition must precede the PyMODINIT_FUNC. */

static void add_uint64_triplet(char **args, const npy_intp *dimensions,
                               const npy_intp *steps, void *data)
{
    npy_intp i;
    npy_intp is1 = steps[0];
    npy_intp is2 = steps[1];
    npy_intp os = steps[2];
    npy_intp n = dimensions[0];
    uint64_t *x, *y, *z;

    char *i1 = args[0];
    char *i2 = args[1];
    char *op = args[2];

    for (i = 0; i < n; i++) {

        x = (uint64_t *)i1;
        y = (uint64_t *)i2;
        z = (uint64_t *)op;

        z[0] = x[0] + y[0];
        z[1] = x[1] + y[1];
        z[2] = x[2] + y[2];

        i1 += is1;
        i2 += is2;
        op += os;
    }
}

static struct PyModuleDef moduledef = {
    PyModuleDef_HEAD_INIT,
    "npufunc",
    NULL,
    -1,
    StructUfuncTestMethods,
    NULL,
    NULL,
    NULL,
    NULL
};

PyMODINIT_FUNC PyInit_npufunc(void)
{
    PyObject *m, *add_triplet, *d;
    PyObject *dtype_dict;
    PyArray_Descr *dtype;
    PyArray_Descr *dtypes[3];

    import_array();
    import_umath();

    m = PyModule_Create(&moduledef);
    if (m == NULL) {
        return NULL;
    }

    /* Create a new ufunc object */
    add_triplet = PyUFunc_FromFuncAndData(NULL, NULL, NULL, 0, 2, 1,
                                          PyUFunc_None, "add_triplet",
                                          "add_triplet_docstring", 0);

    dtype_dict = Py_BuildValue("[(s, s), (s, s), (s, s)]",
                               "f0", "u8", "f1", "u8", "f2", "u8");
    PyArray_DescrConverter(dtype_dict, &dtype);
    Py_DECREF(dtype_dict);

    dtypes[0] = dtype;
    dtypes[1] = dtype;
    dtypes[2] = dtype;

    /* Register ufunc for structured dtype */
    PyUFunc_RegisterLoopForDescr((PyUFuncObject *)add_triplet,
                                 dtype,
                                 &add_uint64_triplet,
                                 dtypes,
                                 NULL);

    d = PyModule_GetDict(m);

    PyDict_SetItemString(d, "add_triplet", add_triplet);
    Py_DECREF(add_triplet);
    return m;
}
